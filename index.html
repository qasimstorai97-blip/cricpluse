function getState(match) { const status = String(match.status || "").trim(); const lowerStatus = status.toLowerCase(); const explicit = String(match.matchState || "").toUpperCase(); // Stumps and breaks take priority over the provider's matchEnded flag. const pausedPattern = /\bstumps?\b|\blunch\b|\btea\b|rain delay|bad light|play suspended|day \d+ close|innings break|match delayed|weather interruption|interval/i; if (pausedPattern.test(lowerStatus)) { return "PAUSED"; } const completedPattern = /\b(won|drawn|tied|abandoned|cancelled|canceled|no result|match complete|match completed|innings victory|conceded|forfeited)\b/i; if (completedPattern.test(lowerStatus)) { return "COMPLETED"; } if (explicit === "COMPLETED" || match.matchEnded === true) { return "COMPLETED"; } if (explicit === "PAUSED") return "PAUSED"; if (explicit === "LIVE" || match.matchStarted === true) { return "LIVE"; } if (explicit === "UPCOMING" || match.matchStarted === false) { return "UPCOMING"; } return "UPCOMING"; }
